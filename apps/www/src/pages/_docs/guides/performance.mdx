# Performance Guide

Form Engine is designed with performance as a first-class concern. This guide covers optimization strategies, best practices, and techniques to ensure your forms perform excellently at scale.

## Performance Principles

Form Engine's performance architecture is built on several key principles:

### 1. Minimal Re-renders

Form Engine uses React Hook Form's optimized rendering strategy with strategic `useWatch` implementation:

```typescript
// ✅ Form Engine automatically optimizes re-renders
function ConditionalField({ field, constraints }) {
  // Only re-renders when watched fields change
  const watchedFields = constraints?.map(c => c.field) || [];
  const watchedValues = useWatch({ name: watchedFields });
  
  const shouldShow = evaluateConstraints(constraints, watchedValues);
  
  return shouldShow ? <FieldComponent field={field} /> : null;
}
```

### 2. Tree-Shakable Architecture

Only import the field components you actually use:

```typescript
// ✅ Production: Import only what you need
import "@/components/form/registries/input.registry";
import "@/components/form/registries/select.registry";
import "@/components/form/registries/checkbox.registry";

// ❌ Development: Import everything (for convenience)
import "@/components/form/registries"; // 50KB+ bundle impact
```

### 3. Lazy Validation

Zod schemas are generated only when needed:

```typescript
// Schema generation is deferred until first validation
const schema = useMemo(() => {
  return autoGenerateSchema ? generateZodSchema(formConfig) : undefined;
}, [formConfig, autoGenerateSchema]);
```

## Bundle Size Optimization

### Field Registry Optimization

Control your bundle size by importing only required field types:

```typescript

// array.registry.ts: ~22KB

// Total: ~30KB instead of 120KB+ for all registries
```

### Dynamic Imports for Large Forms

For forms with many field types, use dynamic imports:

```typescript
// src/components/DynamicFormRenderer.tsx
import { lazy, Suspense } from 'react';

const BasicFormRenderer = lazy(() => 
  import("@/lib/form-engine").then(module => ({ 
    default: module.FormRenderer 
  }))
);

const AdvancedFormRenderer = lazy(async () => {
  // Import advanced field types only when needed
  await import("@/components/form/registries/array.registry");
  await import("@/components/form/registries/file.registry");
  await import("@/components/form/registries/date.registry");
  
  const module = await import("@/lib/form-engine");
  return { default: module.FormRenderer };
});

export function DynamicFormRenderer({ config, ...props }) {
  const hasAdvancedFields = config.steps.some(step =>
    step.elements.some(field => 
      ['array', 'file', 'date'].includes(field.type)
    )
  );

  const Renderer = hasAdvancedFields ? AdvancedFormRenderer : BasicFormRenderer;
  
  return (
    <Suspense fallback={<FormSkeleton />}>
      <Renderer config={config} {...props} />
    </Suspense>
  );
}
```

## Runtime Performance

### Constraint Evaluation Optimization

Optimize conditional logic for large forms:

```typescript
// ✅ Optimized constraint evaluation
const optimizedConstraints = useMemo(() => {
  return constraints?.reduce((acc, constraint) => {
    const key = constraint.field;
    if (!acc[key]) acc[key] = [];
    acc[key].push(constraint);
    return acc;
  }, {} as Record<string, ConstraintLogic[]>);
}, [constraints]);

// Watch only the fields that have constraints
const watchedFields = Object.keys(optimizedConstraints);
const formValues = useWatch({ name: watchedFields });

// Evaluate constraints efficiently
const constraintResults = useMemo(() => {
  return Object.entries(optimizedConstraints).reduce((acc, [field, fieldConstraints]) => {
    acc[field] = evaluateFieldConstraints(fieldConstraints, formValues[field]);
    return acc;
  }, {});
}, [optimizedConstraints, formValues]);
```

```typescript
// Performance monitoring
function PerformanceMonitor({ children }) {
  useEffect(() => {
    const observer = new PerformanceObserver((list) => {
      list.getEntries().forEach((entry) => {
```

## Performance Testing

### Load Testing

Test form performance under load:

```typescript
// Performance test utilities
export class FormPerformanceTester {
  private metrics: PerformanceMetric[] = [];

  async testFormRender(config: FormConfig, iterations = 100) {
    for (let i = 0; i < iterations; i++) {
      const start = performance.now();
      
      render(<FormRenderer config={config} />);
      
      const end = performance.now();
      this.metrics.push({
        operation: 'render',
        duration: end - start,
        iteration: i
      });
    }

    return this.getStats();
  }

  async testValidation(formData: any, schema: ZodSchema, iterations = 1000) {
    for (let i = 0; i < iterations; i++) {
      const start = performance.now();
      
      await schema.parseAsync(formData);
      
      const end = performance.now();
      this.metrics.push({
        operation: 'validation',
        duration: end - start,
        iteration: i
      });
    }

    return this.getStats();
  }

  private getStats() {
    const durations = this.metrics.map(m => m.duration);
    return {
      min: Math.min(...durations),
      max: Math.max(...durations),
      avg: durations.reduce((a, b) => a + b, 0) / durations.length,
      p95: this.percentile(durations, 0.95),
      p99: this.percentile(durations, 0.99)
    };
  }

  private percentile(values: number[], p: number) {
    const sorted = values.sort((a, b) => a - b);
    const index = Math.ceil(sorted.length * p) - 1;
    return sorted[index];
  }
}
```

### Real User Monitoring

Monitor real user performance:

```typescript
// Real User Monitoring (RUM)
function useFormPerformanceMonitoring(formId: string) {
  useEffect(() => {
    const startTime = performance.now();

    return () => {
      const endTime = performance.now();
      const duration = endTime - startTime;

      // Send to analytics service
      analytics.track('form_session_duration', {
        formId,
        duration,
        timestamp: new Date().toISOString()
      });
    };

    analytics.track(`form_${event}`, {
      formId,
      timestamp: new Date().toISOString(),
      ...data
    });
  }, [formId]);

  return { trackInteraction };
}
```

## Common Performance Issues

### Issue 1: Excessive Re-renders

```typescript
// ❌ Problematic: Creates new objects on every render
function ProblematicForm() {
  return (
    <FormRenderer
      config={formConfig}
      onSubmit={(data) => submitForm(data)} // New function every render
      initialValues={{ name: "" }}          // New object every render
    />
  );
}

// ✅ Optimized: Stable references
function OptimizedForm() {
  const handleSubmit = useCallback((data) => submitForm(data), []);
  const initialValues = useMemo(() => ({ name: "" }), []);

  return (
    <FormRenderer
      config={formConfig}
      onSubmit={handleSubmit}
      initialValues={initialValues}
    />
  );
}
```

### Issue 2: Large Bundle Size

```typescript
// ❌ Imports everything
import "@/components/form/registries";

// ✅ Tree-shakable imports
import "@/components/form/registries/input.registry";
import "@/components/form/registries/select.registry";

// ✅ Dynamic imports for conditional features
async function loadAdvancedRegistries() {
  if (needsAdvancedFields) {
    await import("@/components/form/registries/array.registry");
    await import("@/components/form/registries/file.registry");
  }
}
```

### Issue 3: Inefficient Constraint Evaluation

```typescript
// ❌ Evaluates all constraints on every change
function InefficientConstraints({ constraints, formData }) {
  const results = constraints.map(constraint => 
    evaluateConstraint(constraint, formData)
  );
  // Re-evaluates everything when any field changes
}

// ✅ Efficient constraint evaluation
function EfficientConstraints({ constraints, formData }) {
  const results = useMemo(() => {
    return constraints.map(constraint => 
      evaluateConstraint(constraint, formData)
    );
  }, [constraints, formData]); // Only re-evaluate when dependencies change
}
```
