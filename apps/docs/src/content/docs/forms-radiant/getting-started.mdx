# Getting Started

<FormRendererPreview config={contactFormConfig} />

## 1. Install peers

```bash
pnpm add @schema-engine/forms-radiant @schema-engine/forms @ecopages/radiant-ui @ecopages/radiant @ecopages/jsx @ecopages/signals zod
```

Load Radiant UI theme and styles once in your app entry:

```typescript
import '@ecopages/radiant-ui/themes/default';
import '@ecopages/radiant-ui/styles.css';
```

## 2. Create a form engine

Default field registrations ship in the package:

```typescript
import { createRadiantFormEngine, schemaFormRegistrations } from '@schema-engine/forms-radiant';

const formEngine = createRadiantFormEngine({ elements: schemaFormRegistrations });
```

## 3. Configure your form

Use the same `FormConfig` JSON as the React adapter:

For TypeScript authored next to your engine, prefer
[`engine.defineForm()`](/docs/forms/builder#author-form-configuration).
See [Authoring modes](/docs/forms/core-concepts#authoring-modes).

```typescript
import type { FormConfig } from '@schema-engine/forms/schemas';

const formConfig: FormConfig = {
	id: 'contact-form',
	steps: [
		{
			id: 'step-1',
			elements: [
				{
					$type: 'input',
					id: 'name',
					name: 'name',
					label: 'Full Name',
					rules: [{ type: 'required', message: 'Name is required' }],
				},
				{
					$type: 'input',
					id: 'email',
					name: 'email',
					label: 'Email',
					inputType: 'email',
					rules: [
						{ type: 'required', message: 'Email is required' },
						{ type: 'email', message: 'Invalid email' },
					],
				},
				{ $type: 'submit', id: 'submit', name: 'submit', label: 'Send' },
			],
		},
	],
};
```

## 4. Render the form

```typescript
import { FormRenderer } from '@schema-engine/forms-radiant';
import { radiantAdapter } from '@schema-engine/forms-radiant/adapters/radiant';

<FormRenderer
	engine={formEngine}
	adapter={radiantAdapter}
	config={formConfig}
	onSubmit={(data) => console.log(data)}
/>;
```

## Override a registration

```typescript
import { InputElementRegistration } from '@schema-engine/forms-radiant/registrations';

const engine = createRadiantFormEngine({
	elements: [...schemaFormRegistrations, { ...InputElementRegistration, render: MyCustomInput }],
});
```

## Author a custom field widget

Radiant field widgets use `bindField(config, controlOverrides?)` from
`@schema-engine/forms-radiant/primitives` to split field configuration into
chrome layout and static control props.

For standard chrome (label, required marker, description, error), spread
`fieldGroupProps` onto `FieldGroup`:

```tsx
import { FieldGroup, bindField } from '@schema-engine/forms-radiant/primitives';
import { RuiInput } from '@ecopages/radiant-ui/input';

export function MyCustomInput(config) {
	const { fieldGroupProps, controlProps } = bindField(config, {
		type: config.inputType,
		placeholder: config.placeholder,
	});

	return (
		<FieldGroup {...fieldGroupProps}>
			<RuiInput
				{...controlProps}
				on:input={(event: Event) => {
					const target = event.target as HTMLInputElement;
					config.onChange?.(target.value);
				}}
				on:blur={config.onBlur}
				on:focus={config.onFocus}
			/>
		</FieldGroup>
	);
}
```

For custom label arrangements (such as checkboxes, switches, or radios),
spread `hostProps` directly on `FormField`:

```tsx
import { FormField, bindField } from '@schema-engine/forms-radiant/primitives';
import { RuiCheckbox } from '@ecopages/radiant-ui/checkbox';
import { RuiFieldDescription, RuiFieldError } from '@ecopages/radiant-ui/field';

export function MyCustomCheckbox(config) {
	const { hostProps, controlProps } = bindField(config);

	return (
		<FormField {...hostProps}>
			<RuiCheckbox
				{...controlProps}
				on:change={(event: Event) => {
					const detail = (event as CustomEvent<{ checked: boolean }>).detail;
					config.onChange?.(detail?.checked ?? false);
				}}
				on:blur={config.onBlur}
				on:focus={config.onFocus}
			>
				{config.label}
			</RuiCheckbox>
			{config.description ? <RuiFieldDescription>{config.description}</RuiFieldDescription> : null}
			<RuiFieldError />
		</FormField>
	);
}
```

`bindField` automatically resolves validation rules from the owning form engine,
merges author-defined `htmlAttributes` under control props, and serializes
SSR hydration attributes (`data-se-field-rules`, `data-se-validation-rules`).
Event handlers stay explicit on the JSX node.
