# Multi-Step Forms

Form Engine provides powerful support for multi-step forms with built-in navigation, step validation, and customizable stepper components. Multi-step forms are perfect for complex workflows like user registration, surveys, checkout processes, and data collection wizards.

## Basic Multi-Step Setup

Create a multi-step form by defining multiple steps in your configuration:

```typescript
const multiStepFormConfig = {
  id: "user-registration",
  title: "Create Your Account",
  steps: [
    {
      title: "Personal Information",
      description: "Tell us about yourself",
      fields: [
        {
          type: "input",
          name: "firstName",
          label: "First Name",
          validation: [{ type: "required" }]
        },
        {
          type: "input",
          name: "lastName",
          label: "Last Name",
          validation: [{ type: "required" }]
        },
        {
          type: "date",
          name: "birthDate",
          label: "Date of Birth",
          dateType: "date",
          validation: [{ type: "required" }]
        }
      ]
    },
    {
      title: "Account Details",
      description: "Set up your login credentials",
      fields: [
        {
          type: "input",
          name: "email",
          label: "Email Address",
          inputType: "email",
          validation: [
            { type: "required" },
            { type: "email" }
          ]
        },
        {
          type: "input",
          name: "password",
          label: "Password",
          inputType: "password",
          validation: [
            { type: "required" },
            { type: "minLength", value: 8 }
          ]
        },
        {
          type: "input",
          name: "confirmPassword",
          label: "Confirm Password",
          inputType: "password",
          validation: [
            { type: "required" },
            {
              type: "custom",
              message: "Passwords must match",
              validator: (value, formData) => value === formData.password
            }
          ]
        }
      ]
    },
    {
      title: "Preferences",
      description: "Customize your experience",
      fields: [
        {
          type: "checkbox",
          name: "newsletter",
          label: "Subscribe to newsletter",
          description: "Receive updates about new features"
        },
        {
          type: "select",
          name: "theme",
          label: "Preferred Theme",
          options: [
            { value: "light", label: "Light" },
            { value: "dark", label: "Dark" },
            { value: "auto", label: "Auto" }
          ],
          defaultValue: "auto"
        }
      ]
    }
  ]
};
```

## Stepper Configuration

Customize the stepper appearance and behavior:

```typescript
const formWithStepperConfig = {
  id: "advanced-form",
  title: "Advanced Multi-Step Form",
  steps: [/* ... */],
  stepper: {
    variant: "default",              // Stepper visual style
    orientation: "horizontal",       // Layout direction
    showStepNumbers: true,           // Display step numbers
    showStepTitles: true,           // Display step titles
    validateOnNext: true,           // Validate current step before proceeding
    allowSkipping: false,           // Allow jumping to any step
    showProgress: true,             // Show progress indicator
    className: "my-custom-stepper"  // Custom CSS class
  }
};
```

## Step Validation

Control how and when steps are validated:

```typescript
const validatedFormConfig = {
  id: "wizard-form",
  steps: [
    {
      title: "Step 1",
      fields: [
        {
          type: "input",
          name: "email",
          label: "Email",
          validation: [
            { type: "required" },
            { type: "email" }
          ]
        }
      ]
    },
    {
      title: "Step 2", 
      fields: [
        {
          type: "input",
          name: "phone",
          label: "Phone Number",
          validation: [
            { type: "required" },
            {
              type: "custom",
              message: "Please enter a valid phone number",
              validator: (value) => /^\+?[\d\s\-\(\)]+$/.test(value)
            }
          ]
        }
      ]
    }
  ],
  stepper: {
    validateOnNext: true,     // Validate before moving to next step
    validateOnPrevious: false, // Don't validate when going back
    showErrors: true,         // Show validation errors immediately
    blockInvalidSteps: true   // Prevent navigation to invalid steps
  }
};
```

## Manual Stepper Control

For advanced use cases, use the stepper components directly:

```typescript
import { Stepper, FormRenderer } from "@/lib/form-engine";
import { useState } from "react";

function CustomMultiStepForm() {
  const [currentStep, setCurrentStep] = useState(0);
  const [formData, setFormData] = useState({});
  
  const handleStepChange = (newStep: number) => {
    // Custom validation or logic before step change
    if (validateCurrentStep()) {
      setCurrentStep(newStep);
    }
  };

  const handleNext = () => {
    if (currentStep < formConfig.steps.length - 1) {
      setCurrentStep(currentStep + 1);
    }
  };

  const handlePrevious = () => {
    if (currentStep > 0) {
      setCurrentStep(currentStep - 1);
    }
  };

  return (
    <div className="max-w-4xl mx-auto p-6">
      <Stepper 
        sections={formConfig.steps}
        currentStep={currentStep}
        onStepChange={handleStepChange}
      >
        <Stepper.Header 
          variant="default"
          orientation="horizontal"
          showStepNumbers={true}
          className="mb-8"
        />
        
        <Stepper.Content>
          <FormRenderer
            config={{
              ...formConfig,
              steps: [formConfig.steps[currentStep]]
            }}
            onSubmit={handleStepSubmit}
            onChange={setFormData}
            initialValues={formData}
          />
        </Stepper.Content>
        
        <Stepper.Navigation
          previousButton={{
            label: "Back",
            variant: "outline",
            disabled: currentStep === 0
          }}
          nextButton={{
            label: currentStep === formConfig.steps.length - 1 ? "Submit" : "Next",
            variant: "default"
          }}
          onPrevious={handlePrevious}
          onNext={handleNext}
          className="mt-6"
        />
      </Stepper>
    </div>
  );
}
```

## Advanced Stepper Customization

Create fully custom stepper components:

```typescript
import { StepperProvider, useStepperContext } from "@/lib/form-engine";

function CustomStepperHeader() {
  const { currentStep, sections, goToStep } = useStepperContext();
  
  return (
    <div className="flex items-center justify-between mb-8">
      {sections.map((section, index) => (
        <div key={index} className="flex items-center">
          <button
            onClick={() => goToStep(index)}
            className={cn(
              "w-10 h-10 rounded-full flex items-center justify-center",
              index === currentStep 
                ? "bg-blue-600 text-white" 
                : index < currentStep
                ? "bg-green-600 text-white"
                : "bg-gray-200 text-gray-600"
            )}
          >
            {index < currentStep ? "✓" : index + 1}
          </button>
          <span className="ml-2 text-sm font-medium">
            {section.title}
          </span>
          {index < sections.length - 1 && (
            <div className="mx-4 h-px bg-gray-300 flex-1" />
          )}
        </div>
      ))}
    </div>
  );
}

function CustomMultiStepForm() {
  return (
    <StepperProvider sections={formConfig.steps}>
      <CustomStepperHeader />
      <StepperContent>
        {/* Form content */}
      </StepperContent>
      <StepperNavigation />
    </StepperProvider>
  );
}
```
