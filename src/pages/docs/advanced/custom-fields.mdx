# Custom Fields

Form Engine's extensible architecture allows you to create custom field types that integrate seamlessly with the existing validation, conditional logic, and grid systems.

## Creating Custom Field Components

Custom fields are React components that follow the Form Engine field interface:

```typescript
interface CustomFieldProps {
  field: FieldConfig;
  value: any;
  onChange: (value: any) => void;
  error?: string;
  disabled?: boolean;
}

export function CustomSliderField({ field, value, onChange, error, disabled }: CustomFieldProps) {
  return (
    <div className="space-y-2">
      <label className="text-sm font-medium">{field.label}</label>
      <input
        type="range"
        min={field.min || 0}
        max={field.max || 100}
        value={value || 0}
        onChange={(e) => onChange(Number(e.target.value))}
        disabled={disabled}
        className="w-full"
      />
      <div className="text-xs text-gray-500">Value: {value || 0}</div>
      {error && <div className="text-xs text-red-500">{error}</div>}
    </div>
  );
}
```

## Registering Custom Fields

Register your custom field with the Form Engine registry:

```typescript
// src/components/form/custom-fields.ts
import { registerField } from '@/lib/form-engine/registry';
import { CustomSliderField } from './custom-slider-field';

// Register the custom field
registerField('slider', CustomSliderField);
```

## Custom Field Configuration

Define the TypeScript interface for your custom field:

```typescript
interface SliderFieldConfig extends BaseFieldConfig {
  type: 'slider';
  min?: number;
  max?: number;
  step?: number;
}

// Usage in form configuration
{
  type: 'slider',
  id: 'rating',
  name: 'rating',
  label: 'Rate this experience',
  min: 1,
  max: 10,
  step: 1,
  validation: [
    ValidationPresets.required('Please provide a rating'),
    ValidationPresets.min(1, 'Rating must be at least 1'),
    ValidationPresets.max(10, 'Rating cannot exceed 10')
  ]
}
```

## Validation Integration

Custom fields automatically integrate with the validation system:

```typescript
export function CustomColorField({ field, value, onChange, error }: CustomFieldProps) {
  return (
    <div className="space-y-2">
      <label className="text-sm font-medium">{field.label}</label>
      <div className="flex items-center gap-2">
        <input
          type="color"
          value={value || '#000000'}
          onChange={(e) => onChange(e.target.value)}
          className="w-12 h-8 border rounded"
        />
        <input
          type="text"
          value={value || '#000000'}
          onChange={(e) => onChange(e.target.value)}
          placeholder="#000000"
          className="flex-1 px-3 py-2 border rounded"
        />
      </div>
      {error && <div className="text-xs text-red-500">{error}</div>}
    </div>
  );
}

// Validation for color fields
{
  type: 'color',
  name: 'themeColor',
  label: 'Theme Color',
  validation: [
    ValidationPresets.required('Please select a color'),
    ValidationPresets.pattern(/^#[0-9A-Fa-f]{6}$/, 'Please enter a valid hex color')
  ]
}
```

## Advanced Custom Fields

Create complex custom fields with internal state:

```typescript
import { useState } from 'react';

export function CustomTagsField({ field, value, onChange, error }: CustomFieldProps) {
  const [inputValue, setInputValue] = useState('');
  const tags = value || [];

  const addTag = () => {
    if (inputValue.trim() && !tags.includes(inputValue.trim())) {
      onChange([...tags, inputValue.trim()]);
      setInputValue('');
    }
  };

  const removeTag = (tagToRemove: string) => {
    onChange(tags.filter((tag: string) => tag !== tagToRemove));
  };

  return (
    <div className="space-y-2">
      <label className="text-sm font-medium">{field.label}</label>
      <div className="flex flex-wrap gap-2 mb-2">
        {tags.map((tag: string) => (
          <span
            key={tag}
            className="bg-blue-100 text-blue-800 text-xs px-2 py-1 rounded flex items-center gap-1"
          >
            {tag}
            <button
              type="button"
              onClick={() => removeTag(tag)}
              className="text-blue-600 hover:text-blue-800"
            >
              ×
            </button>
          </span>
        ))}
      </div>
      <div className="flex gap-2">
        <input
          type="text"
          value={inputValue}
          onChange={(e) => setInputValue(e.target.value)}
          onKeyPress={(e) => e.key === 'Enter' && (e.preventDefault(), addTag())}
          placeholder="Add a tag..."
          className="flex-1 px-3 py-2 border rounded"
        />
        <button
          type="button"
          onClick={addTag}
          className="px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600"
        >
          Add
        </button>
      </div>
      {error && <div className="text-xs text-red-500">{error}</div>}
    </div>
  );
}
```

## Grid System Integration

Custom fields automatically work with the responsive grid system:

```typescript
{
  type: 'slider',
  name: 'priority',
  label: 'Priority Level',
  grid: {
    responsive: {
      default: { column: '1 / -1' },  // Full width on mobile
      md: { column: '1 / 3' }         // Half width on desktop
    }
  }
}
```

## Conditional Logic Support

Custom fields integrate with the constraint system:

```typescript
{
  type: 'color',
  name: 'customColor',
  label: 'Custom Color',
  constraints: [
    {
      field: 'useCustomColor',
      operator: 'equals',
      value: true,
      action: 'show'
    }
  ]
}
```

## Best Practices

### Accessibility
- Always include proper labels and ARIA attributes
- Support keyboard navigation
- Provide screen reader friendly descriptions

### Performance
- Use React.memo for expensive custom fields
- Debounce onChange calls for real-time validation
- Minimize re-renders with stable references

### Validation
- Always handle edge cases in validation
- Provide clear error messages
- Support both sync and async validation

### Styling
- Use consistent styling with existing fields
- Support both light and dark themes
- Ensure responsive design

This system allows you to extend Form Engine with any custom field type while maintaining full integration with validation, conditional logic, and responsive layouts.
