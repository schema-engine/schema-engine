## Basic Contact Form

A simple contact form with validation and responsive design.

```typescript
import { FormRenderer } from "@/lib/form-engine";

const contactFormConfig = {
  id: "contact-form",
  title: "Get in Touch",
  description: "We'd love to hear from you. Send us a message and we'll respond as soon as possible.",
  steps: [{
    id: "contact-details",
    title: "Contact Information",
    fields: [
      {
        type: "input",
        name: "name",
        label: "Full Name",
        placeholder: "Enter your full name",
        validation: [
          { type: "required", message: "Name is required" },
          { type: "minLength", value: 2, message: "Name must be at least 2 characters" }
        ]
      },
      {
        type: "input",
        name: "email",
        label: "Email Address",
        inputType: "email",
        placeholder: "your.email@example.com",
        validation: [
          { type: "required", message: "Email is required" },
          { type: "email", message: "Please enter a valid email address" }
        ]
      },
      {
        type: "input",
        name: "subject",
        label: "Subject",
        placeholder: "What's this about?",
        validation: [
          { type: "required", message: "Subject is required" }
        ]
      },
      {
        type: "textarea",
        name: "message",
        label: "Message",
        placeholder: "Tell us more about your inquiry...",
        rows: 5,
        validation: [
          { type: "required", message: "Message is required" },
          { type: "minLength", value: 10, message: "Message must be at least 10 characters" }
        ]
      }
    ]
  }],
  layout: {
    columns: 1,
    gap: "1.5rem",
    responsive: {
      default: { columns: 1 },
      md: { columns: 2 }
    }
  }
};

export function ContactForm() {
  const handleSubmit = async (data) => {
    try {
      const response = await fetch("/api/contact", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(data)
      });
      
      if (response.ok) {
        alert("Message sent successfully!");
      } else {
        alert("Failed to send message. Please try again.");
      }
    } catch (error) {
      console.error("Error submitting form:", error);
      alert("An error occurred. Please try again.");
    }
  };

  return (
    <FormRenderer 
      config={contactFormConfig}
      onSubmit={handleSubmit}
    />
  );
}
```

const registrationConfig = {

    <FormRenderer

    />

## Dynamic Survey Form

A survey form with conditional logic and dynamic fields.

```typescript
import { FormRenderer } from "@/lib/form-engine";

const surveyConfig = {
  id: "customer-survey",
  title: "Customer Satisfaction Survey",
  description: "Help us improve our services by sharing your feedback",
  steps: [{
    id: "survey-questions",
    title: "Survey Questions",
    fields: [
      {
        type: "radio",
        name: "satisfaction",
        label: "How satisfied are you with our service?",
        options: [
          { value: "very-satisfied", label: "Very Satisfied" },
          { value: "satisfied", label: "Satisfied" },
          { value: "neutral", label: "Neutral" },
          { value: "dissatisfied", label: "Dissatisfied" },
          { value: "very-dissatisfied", label: "Very Dissatisfied" }
        ],
        validation: [{ type: "required", message: "Please rate your satisfaction" }]
      },
      {
        type: "textarea",
        name: "feedback",
        label: "What could we improve?",
        placeholder: "Please share your thoughts...",
        rows: 4,
        constraints: [
          {
            field: "satisfaction",
            operator: "equals",
            value: "dissatisfied",
            action: "show"
          },
          {
            field: "satisfaction",
            operator: "equals",
            value: "very-dissatisfied",
            action: "show"
          }
        ],
        validation: [
          { type: "required", message: "Please provide feedback for improvement" }
        ]
      },
      {
        type: "radio",
        name: "recommend",
        label: "Would you recommend us to others?",
        options: [
          { value: "definitely", label: "Definitely" },
          { value: "probably", label: "Probably" },
          { value: "maybe", label: "Maybe" },
          { value: "probably-not", label: "Probably Not" },
          { value: "definitely-not", label: "Definitely Not" }
        ],
        validation: [{ type: "required", message: "Please let us know if you'd recommend us" }]
      },
      {
        type: "textarea",
        name: "additional-comments",
        label: "Additional Comments",
        placeholder: "Any other feedback you'd like to share?",
        rows: 3
      },
      {
        type: "checkbox",
        name: "follow-up",
        label: "I'm open to a follow-up discussion about my feedback"
      },
      {
        type: "input",
        name: "contact-email",
        label: "Contact Email",
        inputType: "email",
        placeholder: "your.email@example.com",
        constraints: [
          {
            field: "follow-up",
            operator: "equals",
            value: true,
            action: "show"
          }
        ],
        validation: [
          { type: "required", message: "Email is required for follow-up" },
          { type: "email", message: "Please enter a valid email address" }
        ]
      }
    ]
  }],
  actions: [
    {
      type: "gtm_event",
      trigger: "form_submit",
      eventName: "survey_completed",
      includeFormData: ["satisfaction", "recommend"],
      includeMetadata: true
    }
  ]
};

export function SurveyForm() {
  const handleSubmit = async (data) => {
    // Handle survey submission
    console.log("Survey data:", data);
  };

  return (
    <FormRenderer 
      config={surveyConfig}
      onSubmit={handleSubmit}
    />
  );
}
```
