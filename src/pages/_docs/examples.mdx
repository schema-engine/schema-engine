## Code Examples

### Basic Form Configuration

Here's the configuration that powers the basic demo above:

```typescript
import { FormRenderer } from "@/lib/form-engine";
import { ValidationPresets } from "@/lib/form-engine/validation";
import type { FormConfig } from "@/lib/form-engine/types";

const basicFormConfig: FormConfig = {
  id: "demo-form",
  title: "Interactive Form Demo",
  description: "Try out different field types and configurations",
  autoGenerateSchema: true,
  mode: "free",
  steps: [
    {
      id: "demo-step",
      title: "Demo Form",
      elements: [
        {
          type: "input",
          inputType: "text",
          id: "name",
          name: "name",
          label: "Full Name",
          placeholder: "Enter your full name",
          rules:
            ValidationPresets.minLength(2, "Name must be at least 2 characters"),
            ValidationPresets.maxLength(50, "Name cannot exceed 50 characters"),
          ],
        },
        {
          type: "input",
          inputType: "email",
          id: "email",
          name: "email",
          label: "Email Address",
          placeholder: "Enter your email",
          rules:idationPresets.email("Please enter a valid email address")],
        },
        // ... more fields
      ],
    },
  ],
  actions: [
    {
      id: "demo-form-submit",
      type: "submit",
      description: "Submit demo form",
      successMessage: "Message sent successfully!",
      errorMessage: "Failed to send message. Please try again.",
      enabled: true,
    },
  ],
};

// Usage
export function ContactForm() {
  return <FormRenderer config={basicFormConfig} onSubmit={console.log} />;
}
```

import { FormRenderer } from "@/lib/form-engine";

const contactFormConfig = {

        label: "Email Address",
        inputType: "email",
        placeholder: "your.email@example.com",
        rules:
          { type: "required", message: "Email is required" },
          { type: "email", message: "Please enter a valid email address" }
        ]
      },
      {
        type: "input",

        label: "Subject",

        rules:
          { type: "required", message: "Subject is required" }
        ]
      },
      {
        type: "textarea",
        name: "message",

        rows: 5,
        rules:
          { type: "required", message: "Message is required" },
          { type: "minLength", value: 10, message: "Message must be at least 10 characters" }
        ]
      }
    ]
  }],

    columns: 1,
    gap: "1.5rem",
    responsive: {
      default: { columns: 1 },
      md: { columns: 2 }
    }
  }

export function ContactForm() {
  const handleSubmit = async (data) => {
    try {
      const response = await fetch("/api/contact", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(data)
      });
      
      if (response.ok) {
        alert("Message sent successfully!");
      } else {
        alert("Failed to send message. Please try again.");
      }

      console.error("Error submitting form:", error);
      alert("An error occurred. Please try again.");
